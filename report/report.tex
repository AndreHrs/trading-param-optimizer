\newcommand{\detailtexcount}[1]{%
  \immediate\write18{texcount -merge -sum #1.tex > #1.wcdetail }%
  \verbatiminput{#1.wcdetail}%
}

% Counts : text words, header words
%  Does not count : caption words, #headers, #floats, #inlined formulae,
% #displayed formulae
\newcommand{\quickwordcount}[1]{%
  \immediate\write18{texcount -1 -sum=1,1,0,0,0,0,0 -brief -merge -inc -q
  -relaxed #1.tex | sed "s/ (errors:[0-9]*)//" > #1-words.sum }%
  \input{#1-words.sum} words%
}
